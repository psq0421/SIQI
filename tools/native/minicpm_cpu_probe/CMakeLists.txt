cmake_minimum_required(VERSION 3.22)
project(siqi_minicpm_cpu_probe LANGUAGES C CXX)

# Independent executable and optional opaque-ABI text runtime. Never expose
# this revision's llama/ggml structs through the existing llamadart b10333 ABI.
if(NOT EXISTS "${SIQI_OMNI_SOURCE}/include/llama.h")
    message(FATAL_ERROR "Set SIQI_OMNI_SOURCE to the verified pinned source directory")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
option(SIQI_ENABLE_KLEIDIAI "Use ARM KleidiAI kernels for the Android probe" ON)
set(GGML_CPU_KLEIDIAI ${SIQI_ENABLE_KLEIDIAI} CACHE BOOL "" FORCE)
set(SIQI_ARM_ARCH "armv8.6-a+fp16+dotprod+i8mm" CACHE STRING
    "ARM architecture for the benchmark device; not a universal APK baseline")
set(GGML_CPU_ARM_ARCH "${SIQI_ARM_ARCH}" CACHE STRING "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_CPU_REPACK OFF CACHE BOOL "" FORCE)
option(SIQI_ENABLE_VULKAN "Build the optional Vulkan probe backend" OFF)
set(GGML_VULKAN ${SIQI_ENABLE_VULKAN} CACHE BOOL "" FORCE)
add_subdirectory("${SIQI_OMNI_SOURCE}" omni)
add_executable(siqi-minicpm-cpu-probe main.cpp)
target_compile_features(siqi-minicpm-cpu-probe PRIVATE cxx_std_17)
target_include_directories(siqi-minicpm-cpu-probe PRIVATE "${SIQI_OMNI_SOURCE}/vendor")
target_link_libraries(siqi-minicpm-cpu-probe PRIVATE llama)

option(SIQI_BUILD_LES_LIBRARY "Build SIQI's separate text-only FFI runtime" OFF)
if(SIQI_BUILD_LES_LIBRARY)
    add_library(siqi_les SHARED les_runtime.cpp)
    target_compile_features(siqi_les PRIVATE cxx_std_17)
    target_include_directories(siqi_les PRIVATE "${SIQI_OMNI_SOURCE}/vendor")
    target_link_libraries(siqi_les PRIVATE llama)
    set_target_properties(siqi_les PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
    # Hide all upstream static-library symbols to prevent collisions with the
    # app's different ggml/llamadart ABI. Only our small siqi_les_* ABI is public.
    target_link_options(siqi_les PRIVATE "-Wl,--exclude-libs,ALL")
    add_executable(siqi-les-abi-test les_abi_test.cpp)
    target_link_libraries(siqi-les-abi-test PRIVATE siqi_les)
    target_compile_features(siqi-les-abi-test PRIVATE cxx_std_17)
endif()
